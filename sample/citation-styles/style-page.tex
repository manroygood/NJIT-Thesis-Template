% !TEX root = style-acm.tex
% One page of the citation-style samples in Appendix B of the sample
% dissertation. Each style-*.tex file sets
% \stylename, \stylecode, and \stylefeatures (the items of a bulleted list),
% \stylespecial (Special Instructions, if the style has any),
% \stylecase (what the style does to the capitalization of titles), and then
% reads this file, so every page lists the same sources from
% ../references.bib in the same order. To try another style, copy one of
% the style-*.tex files, change its definitions, build it, and add a
% \stylesample line for it to ../appendixB.tex.

\documentclass[12pt,letterpaper]{article}
\usepackage[left=1.5in,right=1in,top=1in,bottom=1in]{geometry}
\usepackage[T1]{fontenc}
\usepackage{newtxtext,newtxmath}
\usepackage{csquotes}
\usepackage{xurl}
\usepackage{etoolbox}
\usepackage{enumitem}
\usepackage{setspace}

% The same options as the biblatex line in ../../packages.tex; only the style
% changes from page to page. Chicago is the exception: its style must be
% loaded through the biblatex-chicago package, and it spells out given
% names, so giveninits is left out.
\ifdefstring{\stylecode}{chicago-authordate}
  {\usepackage[authordate,backend=biber,doi=false,url=false,%
     isbn=false]{biblatex-chicago}%
   \newcommand{\stylesetup}{\texttt{\textbackslash usepackage[authordate,\ldots]\{biblatex-chicago\}}}}
  {\usepackage[style=\stylecode,%
     backend=biber,doi=false,url=false,defernums=true,sortcites=true,%
     isbn=false,giveninits=true]{biblatex}%
   \newcommand{\stylesetup}{\texttt{style=\stylecode}}}
\addbibresource{../references.bib}
\usepackage[hidelinks]{hyperref}

\pagestyle{empty}
% Laid out like the dissertation, in 12 pt: 1.5 in left margin; headings
% bold, centered, and in capitals; single-spaced bullets and references,
% with a blank line between references; 0.5 in paragraph indent. The text
% is single-spaced so that each style fits on one page.
\setlength{\parindent}{0.5in}
\setlength{\emergencystretch}{2em}
\setlength{\bibitemsep}{\baselineskip}
\defbibheading{bibliography}[\bibname]{%
  \par\vspace{0.25\baselineskip}%
  {\centering\normalsize\bfseries\MakeUppercase{#1}\par}\vspace{0.25\baselineskip}}

\newlength{\stylehead}\setlength{\stylehead}{1.5\baselineskip}
\begin{document}
% The page is inserted into Appendix B of the sample, which prints the
% heading (B.1 ACM, ...) in the space left blank here.
\vspace*{\stylehead}
\noindent\stylesetup

\begin{itemize}[nosep,leftmargin=1.5em,topsep=0.25\baselineskip]
\stylefeatures
\end{itemize}

\par\smallskip\noindent\textbf{In references.bib:} type titles in title case, with braces around names and acronyms (\texttt{\{NumPy\}}), and journal names in full. \stylecase\par

\ifdefined\stylespecial
  \par\smallskip\noindent\textbf{Special Instructions:} \stylespecial\par
\fi

% The same sources in the same order on every page: alphabetical by
% author, which is also the order that numbers them in the styles that
% number sources in order of citation.
\nocite{Adriazola2021,Halmos1973,Harris2020,KnuthPlass1981,Lamport1994,biblatex,Mittelbach2023,EberhardtWiki}

\printbibliography[title={References}]
\end{document}
