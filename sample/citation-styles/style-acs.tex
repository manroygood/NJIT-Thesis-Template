\newcommand{\stylename}{ACS}
\newcommand{\stylecode}{chem-acs}
\newcommand{\stylecase}{This style leaves out article titles.}
\newcommand{\stylefeatures}{%
  \item Citations are numbered in order of first citation: in brackets in the text, [1], or as superscripts (see Special Instructions); the reference list puts the numbers in parentheses.
  \item Family name first, then initials, with semicolons between authors: \emph{Knuth, D.~E.; Plass, M.~F.}
  \item Journal name italic, year bold, volume italic: \emph{Software: Practice and Experience} \textbf{1981}, \emph{11}, 1119--1184.
  \item Books: \emph{Title}, edition; Publisher: City, Year.}
\newcommand{\stylespecial}{%
  Here \texttt{\textbackslash cite} gives numbers in parentheses in the reference list and in brackets in the text. ACS journals print superscripts, which \texttt{\textbackslash autocite} gives: ``Knuth and Plass\textsuperscript{4} showed''''. A superscript is not a word, so never write ``Reference\textsuperscript{4} shows''.}
