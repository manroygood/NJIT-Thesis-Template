% Vancouver. One page of the citation-style samples; see style-page.tex.
\newcommand{\stylename}{Vancouver}
\newcommand{\stylecode}{njit-vancouver}
\newcommand{\stylecase}{This style changes article and book titles to sentence case.}
\newcommand{\stylefeatures}{%
  \item Citations are numbers in order of first citation, in square brackets; the reference list numbers them 1., 2., \ldots
  \item Family name, then initials without periods: Knuth DE, Plass MF. Authors are separated by commas only, with no ``and''.
  \item No title is in italics or quotation marks.
  \item An article ends with year, volume, and shortened page range: \emph{1981; 11:1119--84}.
  \item Books give \emph{City: Publisher, Year}; web pages give \emph{Available from:} the URL and the date you accessed it.}
