\section{Derivation of the Asymptotic Dispersion Relation}
\label{sec:appA_derivation}

Consider a one-dimensional wave packet propagating through a dispersive medium governed by the linearized Klein--Gordon equation:
\begin{equation}
\label{eq:appA_kg}
\frac{\partial^2 \psi}{\partial t^2} - c^2 \frac{\partial^2 \psi}{\partial x^2} + \omega_0^2 \psi = 0.
\end{equation}
Substituting a plane-wave ansatz $\psi(x,t) = A e^{i(kx - \omega t)}$ into Equation~\eqref{eq:appA_kg} yields the algebraic dispersion relation:
\begin{equation}
\label{eq:appA_dispersion}
\omega(k) = \pm \sqrt{c^2 k^2 + \omega_0^2}.
\end{equation}

\begin{proposition}[Group Velocity Bound]
\label{prop:group_vel}
For all wave numbers $k \in \RR$, the group velocity $v_g = \frac{d\omega}{dk}$ is strictly subluminal:
\begin{equation}
|v_g(k)| = \frac{c^2 |k|}{\sqrt{c^2 k^2 + \omega_0^2}} < c.
\end{equation}
\end{proposition}
\begin{proof}
Differentiating Equation~\eqref{eq:appA_dispersion} with respect to $k$ yields $v_g(k) = \frac{c^2 k}{\sqrt{c^2 k^2 + \omega_0^2}}$. Since $\omega_0 > 0$, the denominator satisfies $\sqrt{c^2 k^2 + \omega_0^2} > c|k|$, which immediately implies $|v_g(k)| < c$.
\end{proof}
