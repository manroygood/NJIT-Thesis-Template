\newcommand{\stylename}{AMA}
\newcommand{\stylecode}{njit-ama}
\newcommand{\stylefeatures}{%
  \item Citations are superscript numbers in order of first citation, placed after periods and commas. A page goes in parentheses: \textsuperscript{1(p23)}.
  \item Family name, then initials without periods, with commas between authors: Knuth DE, Plass MF.
  \item Up to six authors are listed; with seven or more, the first three are followed by ``et al.''
  \item Journal and book titles are italic; article titles are in sentence case.
  \item An article ends with year, volume, issue, and pages, with a hyphen: \emph{1981;11(11):1119-1184}. Books give the publisher and year but no city.}
\newcommand{\stylespecial}{%
  Cite with \texttt{\textbackslash autocite}, as this page does, and type it before a period or comma; the number then moves after the punctuation. A superscript is not a word: write ``Knuth and Plass\textsuperscript{4} showed'', never ``Reference\textsuperscript{4} shows''.}
