\newcommand{\stylename}{ASCE}
\newcommand{\stylecode}{njit-asce}
\newcommand{\stylecase}{This style changes article and chapter titles to sentence case.}
\newcommand{\stylefeatures}{%
  \item Citations give the author and year in parentheses: (Knuth and Plass 1981). \texttt{biblatex} has no ASCE style, so the template supplies \texttt{njit-asce}.
  \item Family name, then initials, for every author; the year, in parentheses, follows the authors. Article titles are in quotation marks; book and journal titles are italic.}
\newcommand{\stylespecial}{%
  \texttt{\textbackslash cite} omits the parentheses, so this page cites with \texttt{\textbackslash autocite}; when the authors are part of the sentence, use \texttt{\textbackslash textcite}: ``\textcite{KnuthPlass1981} showed''.}
