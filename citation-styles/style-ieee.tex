% IEEE. One page of the citation-style samples; see style-page.tex.
\newcommand{\stylename}{IEEE}
\newcommand{\stylecode}{ieee}
\newcommand{\stylecase}{This style changes article titles to sentence case.}
\newcommand{\stylefeatures}{%
  \item Citations are numbers in square brackets in order of first citation; a range is written [1]--[3].
  \item Initials come before the family name: D.~E. Knuth.
  \item Article titles are in quotation marks; book and journal titles are italic.
  \item An article ends with \emph{vol.~11, no.~11, pp.~1119--1184, 1981}.
  \item Books give \emph{City, ST: Publisher, Year}; web pages end with \emph{[Online]. Available:} and the URL.}
