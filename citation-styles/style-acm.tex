% ACM. One page of the citation-style samples; see style-page.tex.
\newcommand{\stylename}{ACM}
\newcommand{\stylecode}{njit-acm}
\newcommand{\stylefeatures}{%
  \item Citations are numbers in square brackets, [1]; the reference list is in alphabetical order by author and numbered in that order.
  \item Given names are spelled out (Donald E. Knuth), and the year follows the authors.
  \item Article titles are in sentence case; journal and book titles are italic.
  \item An article ends with volume, issue, year in parentheses, and pages: \emph{11, 11 (1981), 1119--1184}.
  \item Chapters and conference papers: \emph{In Book Title (2nd ed.), Editor (Ed.). Publisher, City, pages.}}
