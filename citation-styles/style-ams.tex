% AMS. One page of the citation-style samples; see style-page.tex.
\newcommand{\stylename}{AMS}
\newcommand{\stylecode}{alphabetic}
\newcommand{\stylecase}{This style prints titles in title case, as you type them.}
\newcommand{\stylefeatures}{%
  \item Citations are labels made from the authors' names and the year, such as [KP81]; the reference list is in alphabetical order.
  \item Initials come before the family name: D.~E. Knuth.
  \item Article titles are in quotation marks; book and journal titles are italic.
  \item An article ends with \emph{In: Journal 11.11 (1981), pp.~1119--1184}.
  \item The AMS's own journals use numbered citations, [1], which \texttt{style=numeric} gives.}
