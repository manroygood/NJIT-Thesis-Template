% Chicago. One page of the citation-style samples; see style-page.tex.
\newcommand{\stylename}{Chicago}
\newcommand{\stylecode}{chicago-authordate}
\newcommand{\stylefeatures}{%
  \item Citations give the author and year with no comma between them: Knuth and Plass 1981. A page follows a comma: Lamport 1994, 23.
  \item Given names are spelled out; only the first author's family name comes first: Knuth, Donald E., and Michael F. Plass.
  \item The year follows the authors. Article titles are in quotation marks and title case; book and journal titles are italic.
  \item An article ends with \emph{11 (11): 1119--1184}; books give \emph{City: Publisher}.
  \item Load it with the \texttt{biblatex-chicago} package, not with \texttt{style=}; see the table of styles in \texttt{packages.tex}.}
