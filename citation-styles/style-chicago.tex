\newcommand{\stylecase}{This style prints titles in title case, as you type them.}
\newcommand{\stylefeatures}{%
  \item Author and year in parentheses, with no comma: (Knuth and Plass 1981).
  \item Given names in full: Knuth, Donald E., and Michael F. Plass. Article titles are in quotation marks; book and journal titles are italic.}
\newcommand{\stylespecial}{%
  \texttt{\textbackslash cite} omits the parentheses, so use \texttt{\textbackslash autocite} for a citation in parentheses; when the authors are part of the sentence, use \texttt{\textbackslash textcite}: ``\textcite{KnuthPlass1981} showed''.}
