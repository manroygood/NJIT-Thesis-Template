% AIP. One page of the citation-style samples; see style-page.tex.
\newcommand{\stylename}{AIP}
\newcommand{\stylecode}{njit-aip}
\newcommand{\stylefeatures}{%
  \item Citations are numbers in square brackets, in order of first citation. Many AIP journals print them as superscripts, which \texttt{\textbackslash autocite} gives.
  \item Initials come before the family name: D.~E. Knuth.
  \item Article titles are in quotation marks and sentence case; journal names are not italic.
  \item An article ends with volume, pages, and the year in parentheses: \emph{11, 1119--1184 (1981)}.
  \item Books put publisher, city, and year together in parentheses.}
