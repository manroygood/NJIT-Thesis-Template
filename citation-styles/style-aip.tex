% AIP. One page of the citation-style samples; see style-page.tex.
\newcommand{\stylename}{AIP}
\newcommand{\stylecode}{njit-aip}
\newcommand{\stylecase}{This style changes article titles to sentence case.}
\newcommand{\stylefeatures}{%
  \item Citations are numbered in order of first citation, in square brackets, [1], or as superscripts (see Special Instructions).
  \item Initials come first (D.~E. Knuth); article titles are in quotation marks; journal names are not italic.
  \item An article ends with \emph{11, 1119--1184 (1981)}; a book ends with \emph{(Publisher, City, Year)}.}
\newcommand{\stylespecial}{%
  Here \texttt{\textbackslash cite} gives numbers in square brackets. Many AIP journals print superscripts, which \texttt{\textbackslash autocite} gives: ``Knuth and Plass\textsuperscript{4} showed''''. A superscript is not a word, so never write ``Reference\textsuperscript{4} shows''.}
