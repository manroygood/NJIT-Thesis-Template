% TRB. One page of the citation-style samples; see style-page.tex.
\newcommand{\stylename}{TRB}
\newcommand{\stylecode}{njit-trb}
\newcommand{\stylecase}{This style changes article and chapter titles to sentence case.}
\newcommand{\stylefeatures}{%
  \item \texttt{biblatex} has no TRB style, so the template supplies \texttt{njit-trb}, built on \texttt{authoryear}. Check your references against the TRB author instructions in Appendix~B of the sample.
  \item Citations give the author and year in parentheses: (Knuth and Plass 1981).
  \item Family name, then initials; the year, in parentheses, follows the authors. Article titles are in quotation marks; book and journal titles are italic.
  \item The reference list is in alphabetical order, with no numbers.}
\newcommand{\stylespecial}{%
  \texttt{\textbackslash cite} omits the parentheses, so use \texttt{\textbackslash autocite} for a citation in parentheses; when the authors are part of the sentence, use \texttt{\textbackslash textcite}: ``\textcite{KnuthPlass1981} showed''.}
