%% Bibliography (BibLaTeX with Biber backend)

% 1. Set the citation style here:

% ==============================================================================
% BIBLATEX CITATION STYLE SELECTION
% Set \mycitestyle to one of the valid style codes below based on your Ph.D. program.
% (No extra \usepackage commands needed)
%
% Citation Style  | \mycitestyle Value   | Style Type
% ------------------------------------------------------------------------------
% ACM             | acm                  | Numeric
% ACS             | chem-acs             | Numeric
% AIP             | chem-aip             | Numeric
% AMA             | ama                  | Numeric (or use 'numeric')
% AMS             | alphabetic           | Alphabetic (or use 'numeric')
% APA             | apa                  | Author-Year
% ASCE            | numeric              | Numeric
% Chicago         | chicago-authordate   | Author-Year
% Harvard         | authoryear           | Author-Year
% IEEE            | ieee                 | Numeric
% TRB             | numeric              | Numeric
% Vancouver       | vancouver            | Numeric
% ==============================================================================
\newcommand{\mycitestyle}{numeric}

% 2. Build and execute the \usepackage command dynamically:
\edef\loadbiblatex{%
  \noexpand\usepackage[style=\mycitestyle,backend=biber,doi=false,url=false,defernums=true,sortcites=true,isbn=false,giveninits=true]{biblatex}%
}
\loadbiblatex
