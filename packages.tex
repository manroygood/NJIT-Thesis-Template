% !TEX root = main.tex
\usepackage[T1]{fontenc}

% Table and alignment packages
% Package tabularx depends on the array package, which is loaded by the class
% file. array also provides the >{\raggedright\arraybackslash} column syntax
% used in the sample tables.
\usepackage{tabularx}
\usepackage{booktabs}

% Math packages
\usepackage{amsmath,amsthm,mathtools}

% Fonts: the GSO requires Times New Roman (or Arial), 12 pt. newtxtext sets
% the text in a Times clone and newtxmath supplies matching math fonts,
% including the AMS symbols (so amssymb/amsfonts must NOT be loaded too).
% Keep newtxmath after amsmath.
\usepackage{newtxtext}
\usepackage{newtxmath}

% Bold math symbols: \bm{p} gives a bold italic p, for vectors. Load bm after
% the math fonts.
\usepackage{bm}

% Graphics and subfigures
\usepackage{graphicx}
\usepackage{subcaption}

% URL breaking for references and tables
\usepackage{xurl}

% Quotations
% This fixes an old-fashioned problem in LaTeX. To get properly formatted
% left and right quotations requires the user to write ``My quote.'' The following
% line allows you to type "My quote," and still get a proper formatting.
\usepackage{csquotes}
\MakeOuterQuote{"}

% Numbers and SI units: \num{12345.678}, \unit{\metre\per\second},
% \qty{3e8}{\metre\per\second}. Requires siunitx version 3 (2021 or later).
\usepackage{siunitx}

% Pseudocode: the algorithmic environment from algpseudocode. Put it in the
% text, like a code listing, not in a figure environment (GSO).
\usepackage{algpseudocode}
\renewcommand{\algorithmicrequire}{\textbf{Input:}}
\renewcommand{\algorithmicensure}{\textbf{Output:}}

% Source code highlighting
% The \color commands in \lstset below depend on the xcolor package, which is
% loaded by the class file.
\usepackage{listings}
\lstset{basicstyle=\ttfamily\small, numbers=none}
\lstset{language=matlab,%
    breaklines=true,%
    keywordstyle=\color{blue},%
    identifierstyle=\color{black},%
    showstringspaces=false,
    emph=[1]{if,for,end,break},emphstyle=[1]\color{red}, %some words to emphasize
    commentstyle=\color{green!40!black}% dark green stays readable when printed
}

%% Bibliography (BibLaTeX with Biber backend)

% 1. Set the citation style here:

% ==============================================================================
% BIBLATEX CITATION STYLE SELECTION
% Set style= in the \usepackage[...]{biblatex} line below to one of the
% style codes in this table, for your discipline (see Table 5.1 in the sample).
%
% Citation Style  | style= value         | Style Type
% ------------------------------------------------------------------------------
% ACM             | acm                  | Numeric
% ACS             | chem-acs             | Numeric
% AIP             | chem-aip             | Numeric
% AMA             | ama                  | Numeric (or use 'numeric')
% AMS             | alphabetic           | Alphabetic (or use 'numeric')
% APA             | apa                  | Author-Year
% ASCE            | numeric              | Numeric
% Chicago         | chicago-authordate   | Author-Year
% Harvard         | authoryear           | Author-Year
% IEEE            | ieee                 | Numeric
% TRB             | numeric              | Numeric
% Vancouver       | vancouver            | Numeric
% ==============================================================================
% TODO (2026-10-15): three codes in the table above are not biblatex styles and
% stop the build with a "not found" error: acm, chem-aip, and ama. Possible
% substitutes: AIP -> phys (biblatex-phys); ACM -> acmnumeric (from the acmart
% package, untested here); AMA has no biblatex style, so vancouver is closest.
% Waiting on the GSO's answer before changing them.

% 2. Load biblatex with that style. Change only the style= value.
\usepackage[style=numeric,% <-- your citation style
  backend=biber,doi=false,url=false,defernums=true,sortcites=true,%
  isbn=false,giveninits=true]{biblatex}


\defbibheading{subbibliography}[\refname]{%
	{\normalfont\large\bfseries}%
}

% Clickable links. hyperref turns every cross-reference, citation, and Table
% of Contents entry into a link, and adds bookmarks to the PDF. The option
% hidelinks keeps the links black, without colored boxes, so the printed
% copy looks the same. Load hyperref after all other packages except cleveref.
\usepackage[hidelinks]{hyperref}

% Cross-references that name what they point to: \cref{fig:eberhardt} prints
% "Figure 4.1". Must be loaded after hyperref.
\usepackage[capitalise,noabbrev]{cleveref}
\crefname{subsection}{Subsection}{Subsections}
% Sections inside an appendix are still "Section C.1", not "Appendix C.1"
\crefname{subappendix}{Section}{Sections}
\crefname{subsubappendix}{Subsection}{Subsections}
