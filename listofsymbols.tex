% List of Symbols (included via \symbols{\input{listofsymbols}} in main.tex)

\begin{description}
\item[$\%$] Percentage sign
\item[$H^1(\RR)$] The Sobolev space of functions over $\RR$ that are square-integrable and whose weak first derivative is square-integrable
\item[$\mathrm{GL}_2(\RR)$] The general linear group of real invertible $2\times 2$ matrices
\item[$E$] Energy of a physical system
\item[$m$] Mass of a physical body
\item[$c$] Speed of light in vacuum ($c \approx 3 \times 10^8\text{ m/s}$)
\item[$\CC$] The field of complex numbers
\item[$\RR$] The field of real numbers
\item[$\ZZ$] The ring of integers
\end{description}
